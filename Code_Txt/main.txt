                    self.model = model
                    self.policy_net = model.policy

                    action, _ = self.model.predict(obs, deterministic=deterministic)

                    self.model.save(path)

                    self.model = PPO.load(path)

def check_dependencies():
    """检查依赖包是否安装"""
    required_packages = [
        'stable_baselines3',
        'torch',
        'numpy',
        'matplotlib',
        'scipy',
        'cvxpy',
        'gymnasium'
    ]
    
    missing_packages = []
    for package in required_packages:
        try:
            __import__(package.replace('-', '_'))
        except ImportError:
            missing_packages.append(package)
    
    if missing_packages:
        print("❌ 缺少依赖包:")
        for package in missing_packages:
            print(f"  - {package}")
        print("\n请运行以下命令安装:")
        print("pip install stable-baselines3[extra] torch numpy matplotlib scipy cvxpy gymnasium")
        return False
    
    print("✅ 所有依赖包已安装")
    return True

if __name__ == "__main__":
    # 检查依赖
    if not check_dependencies():
        exit(1)
    
    # 运行主程序
    main()
